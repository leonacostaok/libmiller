cmake_minimum_required(VERSION 3.16)
project(libmiller VERSION 0.2.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GMP REQUIRED IMPORTED_TARGET gmp)
option(BUILD_SHARED_LIBS "Build shared library" ON)
option(MILLER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(MILLER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(miller src/miller.c src/decimal.c)
target_compile_features(miller PUBLIC c_std_99)
target_include_directories(miller PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(miller PUBLIC PkgConfig::GMP)
set_target_properties(miller PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(miller PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
add_executable(miller-check examples/check.c)
target_link_libraries(miller-check PRIVATE miller)
add_executable(miller-bench bench/bench.c)
target_link_libraries(miller-bench PRIVATE miller)
if(BUILD_TESTING)
  add_executable(miller-tests tests/test_miller.c)
  target_link_libraries(miller-tests PRIVATE miller)
  add_test(NAME correctness COMMAND miller-tests)
endif()
configure_file(libmiller.pc.in libmiller.pc @ONLY)
install(TARGETS miller EXPORT millerTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/miller.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libmiller.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
